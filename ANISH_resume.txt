DROP TABLE IF EXISTS ENROLLS;
DROP TABLE IF EXISTS STUDENT_PHONE;
DROP TABLE IF EXISTS SECTION;
DROP TABLE IF EXISTS STUDENT;
DROP TABLE IF EXISTS COURSE;
DROP TABLE IF EXISTS DEPARTMENT;CREATE TABLE DEPARTMENT (
    Dept_ID VARCHAR(10) PRIMARY KEY,
    Dept_Name VARCHAR(100) NOT NULL
);

CREATE TABLE COURSE (
    Course_ID VARCHAR(10) PRIMARY KEY,
    Title VARCHAR(100) NOT NULL,
    Credits INT NOT NULL,
    Dept_ID VARCHAR(10),
    FOREIGN KEY (Dept_ID) REFERENCES DEPARTMENT(Dept_ID)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);

CREATE TABLE SECTION (
    Course_ID VARCHAR(10),
    Sec_No INT,
    Semester VARCHAR(10) NOT NULL,
    Year INT NOT NULL,
    PRIMARY KEY (Course_ID, Sec_No),
    FOREIGN KEY (Course_ID) REFERENCES COURSE(Course_ID)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE STUDENT (
    Student_ID VARCHAR(10) PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    DOB DATE
);

CREATE TABLE STUDENT_PHONE (
    Student_ID VARCHAR(10),
    Phone_Number VARCHAR(15),
    PRIMARY KEY (Student_ID, Phone_Number),
    FOREIGN KEY (Student_ID) REFERENCES STUDENT(Student_ID)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE ENROLLS (
    Student_ID VARCHAR(10),
    Course_ID VARCHAR(10),
    Sec_No INT,
    Grade VARCHAR(2),
    PRIMARY KEY (Student_ID, Course_ID, Sec_No),
    FOREIGN KEY (Student_ID) REFERENCES STUDENT(Student_ID)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    FOREIGN KEY (Course_ID, Sec_No) REFERENCES SECTION(Course_ID, Sec_No)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);
